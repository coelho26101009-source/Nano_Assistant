━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  O HELIOS pede SEMPRE confirmação antes de:

  🔴 PERIGO ALTO (sempre pede):
    • Apagar ficheiros permanentemente
    • Formatar drives
    • Escrever no registo do Windows
    • Matar processos do sistema

  🟡 PERIGO MÉDIO (pede se o conteúdo for sensível):
    • Comandos PowerShell com: Remove-Item, Format, del, rd,
      rmdir, Stop-Process, shutdown, Restart-Computer, Clear-Disk
    • Submeter formulários web (comprar, pagar, confirmar, apagar)
    • Mover ficheiros em massa

  Como responder à confirmação:
    → Aparece um modal na interface com CONFIRMAR / CANCELAR
    → Tens 60 segundos para responder
    → Se não responderes, a ação é cancelada automaticamente

  🔊 ESCUTA CONTÍNUA (WAKE-WORD)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Com wake_word_enabled: true o HELIOS escuta sempre em background e acorda
  quando dizes a palavra de activação — sem clicar em nada.

  Instalar um dos motores (leves, ~1% de CPU):
    pip install pvporcupine     (Picovoice — grátis para uso pessoal,
                                 precisa de PICOVOICE_ACCESS_KEY no .env)
    pip install openwakeword    (100% offline, sem chave)

  settings.yaml → voice:
    wake_word_enabled: true
    wake_word_phrase: "hey jarvis"   (frases já incluídas:
                                      porcupine → jarvis, computer, alexa…
                                      openwakeword → hey jarvis, hey mycroft, alexa)
    wake_word_sensitivity: 0.6
    wake_word_backend: auto | porcupine | openwakeword
    wake_word_keyword_path: ""       (.ppn/.onnx próprio para dizeres mesmo "Helios")

  Para a palavra "Helios" precisas de um modelo próprio:
    → Porcupine: cria o .ppn em console.picovoice.ai (grátis)
    → openWakeWord: treina um .onnx (github.com/dscripka/openWakeWord)

  🪟 ARRANCAR COM O WINDOWS (sempre ligado, em background)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Clica com o botão direito no ícone do tray → "🪟 Iniciar com o Windows"
  • Nesse arranque a janela fica escondida: só o ícone no tray
  • Clica (ou duplo clique) no tray para abrir a janela
  • Só corre uma instância — abrir de novo traz a janela existente para a frente
  • Para um PC modesto: ollama_enabled: false (só nuvem, zero carga local)
    ou um modelo local minúsculo, ex. ollama_model: "qwen2.5:1.5b"

  💾 MEMÓRIA PERSISTENTE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  O HELIOS carrega o histórico no arranque — continua a conversa onde ficou,
  mesmo depois de reiniciares o PC.

  O que é injectado em cada resposta:
    1. As últimas mensagens (memory.history_messages, truncadas para caber
       na janela de contexto — memory.max_history_chars)
    2. Os factos que já sabe sobre ti (remember_fact)
    3. Excertos relevantes dos PDFs indexados (RAG, se chromadb instalado)

  Base de dados: data/helios.db (SQLite) | Vetores: data/chroma

  Para limpar o histórico:
    → Clica no botão "Clear" no painel de conversa
    → Ou diz: "Limpa o histórico de conversa"

  ⚙️  CONFIGURAÇÃO  (config/settings.yaml)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Ficheiro .env (variáveis privadas):
    GROQ_API_KEY=...          → Chave API do Groq (obrigatória)
    ELEVENLABS_API_KEY=...    → Chave ElevenLabs (opcional, voz humana)
    OPENAI_API_KEY=...        → Chave OpenAI (opcional)

  Ficheiro .env (variáveis privadas, continuação):
    PICOVOICE_ACCESS_KEY=...  → Chave Picovoice (opcional, wake-word Porcupine)

  settings.yaml (comportamento):
    groq_model: "llama-3.1-8b-instant"   → Modelo de IA (rápido)
    ollama_enabled: true/false            → Modelo local de fallback
    ollama_model: "qwen2.5:1.5b"          → Modelo leve para PCs modestos
    memory.history_messages: 20           → Mensagens recuperadas no arranque
    memory.rag_enabled: true              → Junta excertos dos teus PDFs
    voice.wake_word_enabled: true         → Escuta contínua
    voice.stt_provider: faster_whisper    → STT local e privado
    monitor.cpu_threshold: 90             → Limites dos alertas proactivos
    tts_enabled: true/false               → Liga/desliga voz
    tts_provider: edge/elevenlabs/openai  → Motor de voz
    edge_voice: "pt-PT-DuarteNeural"      → Voz portuguesa masculina
    edge_voice: "pt-PT-RaquelNeural"      → Voz portuguesa feminina

  🔌 COMO TRANSPORTAR O HELIOS (pen/portátil)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Para levar o HELIOS numa pen ou noutro PC:

  1. Copia a pasta HELIOS_V7 para a pen
  2. No PC de destino, garante que tens:
     → Python 3.11+ instalado (python.org)
     → Google Chrome instalado
  3. Abre o terminal na pasta HELIOS_V7
  4. Instala dependências: pip install -r requirements.txt
  5. Corre: .\start.bat

  Para criar um .exe portátil (sem precisar de instalar nada):
  → Corre no terminal: pip install pyinstaller
  → Corre: pyinstaller --onefile --noconsole core\main.py

  🐞 RESOLUÇÃO DE PROBLEMAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  "Não responde / demora muito"
    → Verifica ligação à internet (Groq precisa de internet)
    → Verifica se a API key está correta no .env
    → Vê o log: helios.log

  "Rate limit atingido"
    → Aguarda 30-60 segundos e tenta de novo
    → O modelo llama-3.1-8b-instant tem limites maiores

  "Sem voz"
    → Verifica se tts_enabled: true no settings.yaml
    → Instala: pip install edge-tts pygame

  "Erro no plugin X"
    → Vê detalhes em: helios.log
    → Playwright (web_vision): playwright install chromium

  "A wake-word não reage"
    → Confirma wake_word_enabled: true e que instalaste pvporcupine ou openwakeword
    → Vê o helios.log: diz sempre que backend arrancou ou porque falhou
    → A frase tem de ter modelo: usa "hey jarvis" ou define wake_word_keyword_path

  "Não se lembra de nada"
    → Confirma que data/helios.db existe e não está a ser apagado
    → memory.history_messages > 0 no settings.yaml

  Log completo: HELIOS_V7\helios.log
